### Moment

**File**: `app/Models/Moment.php`

```php
class Moment extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'user_id', 'name', 'description', 'color', 'icon',
        'identity_statement', 'is_active', 'sort_order',
    ];

    protected $casts = [
        'is_active' => 'boolean',
    ];

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function schedule(): HasOne
    {
        return $this->hasOne(MomentSchedule::class);
    }

    public function cue(): HasOne
    {
        return $this->hasOne(Cue::class);
    }

    public function reward(): HasOne
    {
        return $this->hasOne(Reward::class);
    }

    public function instances(): HasMany
    {
        return $this->hasMany(MomentInstance::class);
    }

    // Check if moment is scheduled for a given date
    public function isScheduledFor(Carbon $date): bool
    {
        if (!$this->schedule) return true; // No schedule = every day

        return match ($this->schedule->frequency) {
            'daily' => true,
            'weekly' => in_array($date->dayOfWeekIso, $this->schedule->days_of_week ?? []),
            'custom' => in_array($date->dayOfWeekIso, $this->schedule->days_of_week ?? []),
            default => true,
        };
    }
}
```

### MomentInstance

**File**: `app/Models/MomentInstance.php`

```php
class MomentInstance extends Model
{
    protected $fillable = ['moment_id', 'date', 'completed_at', 'notes'];

    protected $casts = [
        'date' => 'date',
        'completed_at' => 'datetime',
    ];

    public function moment(): BelongsTo
    {
        return $this->belongsTo(Moment::class);
    }

    public function isCompleted(): bool
    {
        return $this->completed_at !== null;
    }

    // Toggle completion
    public function toggle(): void
    {
        $this->completed_at = $this->completed_at ? null : now();
        $this->save();
    }
}
```

## Key Query Patterns

### Daily View — Get today's moments with check-off status

```php
$moments = auth()->user()->moments()
    ->where('is_active', true)
    ->with(['schedule', 'instances' => fn ($q) => $q->whereDate('date', today())])
    ->orderBy('sort_order')
    ->get();
```

### Weekly View — Get week grid (M–S) with completion data

```php
$startOfWeek = now()->startOfWeek(Carbon::MONDAY);
$endOfWeek   = now()->endOfWeek(Carbon::SUNDAY);

$moments = auth()->user()->moments()
    ->where('is_active', true)
    ->with(['schedule', 'instances' => fn ($q) =>
        $q->whereBetween('date', [$startOfWeek, $endOfWeek])
    ])
    ->orderBy('sort_order')
    ->get();

// Frontend pivots instances by day-of-week into M T W T F S S grid
```

### Toggle a Moment Instance (check/uncheck)

```php
$instance = MomentInstance::firstOrCreate(
    ['moment_id' => $momentId, 'date' => $date],
);
$instance->toggle();
```

### Streak Calculation

```php
// Count consecutive completed days backwards from today
$streak = 0;
$date = today();

while (true) {
    $instance = MomentInstance::where('moment_id', $momentId)
        ->whereDate('date', $date)
        ->whereNotNull('completed_at')
        ->exists();

    if (!$instance) break;

    $streak++;
    $date = $date->subDay();
}
```

### Dashboard — Monthly completion percentage

```php
$total = MomentInstance::where('moment_id', $momentId)
    ->whereMonth('date', now()->month)
    ->count();

$completed = MomentInstance::where('moment_id', $momentId)
    ->whereMonth('date', now()->month)
    ->whereNotNull('completed_at')
    ->count();

$percentage = $total > 0 ? round(($completed / $total) * 100) : 0;
```
